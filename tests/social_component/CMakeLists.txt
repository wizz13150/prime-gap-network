cmake_minimum_required(VERSION 3.22)
project(PGNSocialComponent LANGUAGES CXX)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
enable_testing()
add_executable(pgn-social-view-test ../social_view_test.cpp ../../src/ui/social_view.cpp ../../src/ui/social_access_dialog.cpp ../../src/ui/social_forum_thread.cpp ../../assets/pgn.qrc)
target_include_directories(pgn-social-view-test PRIVATE ../../src)
target_compile_features(pgn-social-view-test PRIVATE cxx_std_20)
target_link_libraries(pgn-social-view-test PRIVATE Qt6::Widgets)
add_test(NAME social-view COMMAND pgn-social-view-test)
set_tests_properties(social-view PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
